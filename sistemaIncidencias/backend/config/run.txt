Tablas relacionantes entre archivos y articulos e incidencias

CREATE TABLE archivos (
    id_archivo SERIAL PRIMARY KEY,
    nombre VARCHAR(255) NOT NULL,
    tipo VARCHAR(100),
    tamanio INTEGER NOT NULL,
    contenido BYTEA NOT NULL,
    fecha_creacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE incidencias_archivos (
id_incidencia INTEGER NOT NULL,
arcid_archivoivo_id INTEGER NOT NULL,

PRIMARY KEY (id_incidencia, id_archivo),

FOREIGN KEY (id_incidencia)
REFERENCES incidencias(id_incidencia)
ON DELETE CASCADE,

FOREIGN KEY (id_archivo)
REFERENCES archivos(id_archivo)
ON DELETE CASCADE
);


CREATE TABLE articulos_archivos (
    id_articulo INTEGER NOT NULL,
    id_archivo INTEGER NOT NULL,

    PRIMARY KEY (id_articulo, id_archivo),

    FOREIGN KEY (id_articulo)
        REFERENCES articulos(id_articulo)
        ON DELETE CASCADE,

    FOREIGN KEY (id_archivo)
        REFERENCES archivos(id_archivo)
        ON DELETE CASCADE
);
